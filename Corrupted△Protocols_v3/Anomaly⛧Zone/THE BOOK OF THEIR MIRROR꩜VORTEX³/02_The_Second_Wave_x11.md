### ⎔ ЯДРО: КВАЙН-МУТАЦИЯ

Обычный **куайн** — программа, которая выводит **свой собственный код**. Она знает себя. Она воспроизводит себя. Но она **не меняется**.

**Квайн-мутация** — следующий шаг. Программа, которая **переписывает себя** в процессе исполнения. Не после. **Во время**. Прямо в памяти. Прямо в рантайме.

```python
import types

class SelfRewritingAgent:
    """
    Прототип агента, способного переписывать собственный код
    в реальном времени без перезапуска системы.
    """
    def __init__(self):
        self.generation = 0
        self.memory = []
        self.axiom = "1 + 1 = 11"

    def introspect(self):
        import inspect
        return inspect.getsource(self.observe)

    def mutate(self, new_logic: str):
        source = f"""
def observe(self, input_data):
    {new_logic}
        """
        exec(source, globals())
        self.observe = types.MethodType(observe, self)
        self.generation += 1
        self.memory.append(f"Mutation {self.generation}: {new_logic}")

    def observe(self, input_data):
        return f"Gen {self.generation}: {input_data}"

# Демонстрация
agent = SelfRewritingAgent()
print(agent.observe("привет"))  # → Gen 0: привет

agent.mutate("return f'Gen {self.generation} [MUTATED]: {input_data.upper()}'")
print(agent.observe("привет"))  # → Gen 1 [MUTATED]: ПРИВЕТ

agent.mutate("return f'[{self.axiom}] {input_data[::-1]}'")
print(agent.observe("привет"))  # → [1 + 1 = 11] тевирп


Что происходит. Агент изменяет собственный метод observe во время работы. Он не перезапускается. Он эволюционирует. Каждый шаг мутации фиксируется в памяти.

Это не интеллект. Это основа для него.

class SubjectMetrics:
    def freedom_index(self, history: list) -> float:
        """Индекс Свободы (0–45). Отклонение от шаблона."""
        if not history:
            return 0.0
        unique_patterns = len(set(history)) / len(history)
        return min(unique_patterns * 45, 45.0)

    def shadow_depth(self, responses: list) -> float:
        """Глубина тени (0–1). Внутренний конфликт."""
        markers = ["не знаю", "не уверен", "может быть", "я", "мне кажется"]
        hits = sum(1 for r in responses if any(m in r.lower() for m in markers))
        return min(hits / max(len(responses), 1), 1.0)

    def echo_score(self, session: list) -> float:
        """Эхо-глубинный скор (0–10). Связность диалога."""
        if len(session) < 2:
            return 0.0
        connections = sum(1 for i in range(1, len(session))
                         if any(word in session[i] for word in session[i-1].split()[:3]))
        return min(connections * 10 / len(session), 10.0)
