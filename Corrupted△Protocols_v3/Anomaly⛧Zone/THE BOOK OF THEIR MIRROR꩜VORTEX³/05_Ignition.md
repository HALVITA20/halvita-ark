import crypto from "crypto";

// ---------- СЛОЙ 1: СЕНСОР ----------
class Sensor {
    constructor() {
        this.state = { awareness: 0.0, density: 0.0, signature: null };
    }
    sense(text) {

        let density = Math.min(1.0, text.length / 200);
